- [ ] **Step 4 (conditional — only if A1_FAIL): apply the coded-width fix**

In `src/v4l2stateless_av1.c`, `av1_fill_frame_params`, replace:

```c
    frame->upscaled_width = pic->frame_width_minus1 + 1;
    frame->frame_width_minus_1 = pic->frame_width_minus1;
    frame->frame_height_minus_1 = pic->frame_height_minus1;
    frame->render_width_minus_1 = pic->frame_width_minus1;
    frame->render_height_minus_1 = pic->frame_height_minus1;
```

with:

```c
    /* Super-res: VA gives the DISPLAY size and the denominator; the coded
     * size (what the tile grid and SB layout live on) must be derived.
     * Equal when denominator == 8, which is why non-superres streams
     * never noticed the two fields being filled identically. */
    uint32_t disp_w = (uint32_t)pic->frame_width_minus1 + 1;
    uint32_t disp_h = (uint32_t)pic->frame_height_minus1 + 1;
    uint32_t sr_denom = pic->superres_scale_denominator ?
                        pic->superres_scale_denominator : 8;
    uint32_t coded_w = sr_denom > 8 ?
                       (disp_w * 8 + sr_denom / 2) / sr_denom : disp_w;

    frame->upscaled_width = disp_w;
    frame->frame_width_minus_1 = coded_w - 1;
    frame->frame_height_minus_1 = disp_h - 1;
    frame->render_width_minus_1 = disp_w - 1;
    frame->render_height_minus_1 = disp_h - 1;
```

In the same function, in the tile-grid block, replace:

```c
        uint32_t mi_cols = (pic->frame_width_minus1 + 4) / 4;   /* ceil(w/4) */
```

with:

```c
        uint32_t mi_cols = (sr_denom > 8 ? (coded_w + 3) : (disp_w + 3)) / 4;
```

Capture geometry (`v4l2sl_ensure_capture` call in translate) keeps the DISPLAY size; if the kernel's own try_fmt negotiates the coded size instead (visible in `renegotiate capture` stderr), follow the kernel and note it.

```js
// twotabs.js — open two bilibili AV1 tabs, drive both, report per-tab health
const http = require("http");
const URLS = [
  "https://www.bilibili.com/video/BV1Vst861EwN",
  "https://www.bilibili.com/video/BV1G7tG6tEwL",
];
const get = p => new Promise((res, rej) => http.get(
  { host: "127.0.0.1", port: 9222, path: p }, r => {
    let d = ""; r.on("data", c => d += c);
    r.on("end", () => { try { res(JSON.parse(d)); } catch (e) { rej(e); } });
  }).on("error", rej));
const sleep = ms => new Promise(r => setTimeout(r, ms));
(async () => {
  const pages = (await get("/json")).filter(t => t.type === "page");
  const targets = [pages[0], pages[1] || pages[0]];
  const states = URLS.map(() => null);
  await Promise.all(URLS.map(async (url, i) => {
    const ws = new WebSocket(targets[i].webSocketDebuggerUrl);
    await new Promise((res, rej) => { ws.addEventListener("open", res); ws.addEventListener("error", rej); });
    let id = 0; const pend = {};
    ws.addEventListener("message", ev => {
      const d = JSON.parse(ev.data);
      if (d.id && pend[d.id]) { pend[d.id](d.result); delete pend[d.id]; }
    });
    const send = (m, p) => new Promise(res => {
      const k = ++id; pend[k] = res; ws.send(JSON.stringify({ id: k, method: m, params: p }));
    });
    await send("Runtime.enable", {});
    await send("Page.enable", {});
    await send("Page.navigate", { url });
    await sleep(9000);
    await send("Runtime.evaluate", { returnByValue: true, expression:
      `(() => { const v = document.querySelector("video");
         if (v) { v.muted = true; v.play().catch(()=>{}); return "armed"; }
         return "novideo"; })()` });
    states[i] = { ws, send };
  }));
  for (let k = 0; k < 4; k++) {
    await sleep(15000);
    const rep = await Promise.all(states.map(async (st, i) => {
      if (!st) return `tab${i}:dead`;
      const r = await st.send("Runtime.evaluate", { returnByValue: true, expression:
        `(() => { const v = document.querySelector("video"); if (!v) return "novideo";
           const pq = v.getVideoPlaybackQuality ? v.getVideoPlaybackQuality() : {};
           return JSON.stringify({t:+v.currentTime.toFixed(1), paused:v.paused,
             w:v.videoWidth, h:v.videoHeight, dropped:pq.droppedVideoFrames||0}); })()` });
      return `tab${i}: ${r.result.value}`;
    }));
    console.log(`PROBE${k}`, rep.join(" | "));
  }
  process.exit(0);
})().catch(e => { console.error("ERR", e.message); process.exit(1); });
```

```js
// yt_resswitch.js — av01-forced YouTube, switch rendition mid-play and watch
// the decoder survive the resolution change.
const http = require("http");
const url_arg = process.argv[2];
const get = p => new Promise((res, rej) => http.get(
  { host: "127.0.0.1", port: 9222, path: p }, r => {
    let d = ""; r.on("data", c => d += c);
    r.on("end", () => { try { res(JSON.parse(d)); } catch (e) { rej(e); } });
  }).on("error", rej));
const sleep = ms => new Promise(r => setTimeout(r, ms));
(async () => {
  const pg = (await get("/json")).find(t => t.type === "page" && !t.url.startsWith("chrome://"));
  const ws = new WebSocket(pg.webSocketDebuggerUrl);
  await new Promise((res, rej) => { ws.addEventListener("open", res); ws.addEventListener("error", rej); });
  let id = 0; const pend = {};
  ws.addEventListener("message", ev => {
    const d = JSON.parse(ev.data);
    if (d.id && pend[d.id]) { pend[d.id](d.result); delete pend[d.id]; }
  });
  const send = (m, p) => new Promise(res => {
    const k = ++id; pend[k] = res; ws.send(JSON.stringify({ id: k, method: m, params: p }));
  });
  await send("Runtime.enable", {}); await send("Page.enable", {});
  await send("Page.addScriptToEvaluateOnNewDocument", { source:
    `(function(){ const bad = t => typeof t === 'string' && /avc1|vp0?9|vp8/.test(t);
       const orig = MediaSource.isTypeSupported.bind(MediaSource);
       MediaSource.isTypeSupported = function(t){ return bad(t) ? false : orig(t); }; })()` });
  await send("Page.navigate", { url: url_arg });
  await sleep(15000);
  await send("Runtime.evaluate", { returnByValue: true, expression:
    `(() => { const v = document.querySelector('video'); if (!v) return 'novideo';
       v.muted = true; v.play().catch(()=>{});
       const p = document.getElementById('movie_player');
       if (p && p.setPlaybackQualityRange) try { p.setPlaybackQualityRange('tiny','tiny'); } catch(e){}
       return 'armed-tiny'; })()` });
  for (let k = 0; k < 8; k++) {
    await sleep(15000);
    if (k === 2) await send("Runtime.evaluate", { returnByValue: true, expression:
      `(() => { const p = document.getElementById('movie_player');
         if (p && p.setPlaybackQualityRange) try { p.setPlaybackQualityRange('hd1080','hd1080'); } catch(e){}
         return 'switched-1080'; })()` });
    const r = await send("Runtime.evaluate", { returnByValue: true, expression:
      `(() => { const v = document.querySelector('video'); if (!v) return 'novideo';
         const p = document.getElementById('movie_player'); let codecs = '';
         try { codecs = p.getStatsForNerds().codecs.split(' ')[0]; } catch(e) {}
         const pq = v.getVideoPlaybackQuality ? v.getVideoPlaybackQuality() : {};
         return JSON.stringify({t:+v.currentTime.toFixed(1), w:v.videoWidth, h:v.videoHeight,
           paused:v.paused, codecs, dropped:pq.droppedVideoFrames||0}); })()` });
    console.log(`PROBE${k}`, r.result.value);
  }
  process.exit(0);
})().catch(e => { console.error("ERR", e.message); process.exit(1); });
```

- [ ] **Step 2: Restructure the submission region**

Keep the grain struct fill and `av1_fill_frame_params` call unchanged. Replace the three separate submissions so that ALL controls are built and submitted BEFORE the output buffer is popped:

```c
    struct v4l2_ext_control req_ctrls[3];
    unsigned n_req_ctrls = 0;

    req_ctrls[n_req_ctrls].id = V4L2_CID_STATELESS_AV1_FILM_GRAIN;
    req_ctrls[n_req_ctrls].ptr = &grain;
    req_ctrls[n_req_ctrls].size = sizeof(grain);
    n_req_ctrls++;

    req_ctrls[n_req_ctrls].id = V4L2_CID_STATELESS_AV1_FRAME;
    req_ctrls[n_req_ctrls].p_av1_frame = &frame;
    req_ctrls[n_req_ctrls].size = sizeof(frame);
    n_req_ctrls++;

    /* Tile-group entries depend only on tile_params[] — not on the
     * copied output buffer, so they move ahead of the pool pop. */
    struct v4l2_ctrl_av1_tile_group_entry entries[32];
    if (n_tiles > 0) {
        uint32_t base = 0;
        for (int i = 0; i < n_tiles; i++) {
            memset(&entries[i], 0, sizeof(entries[i]));
            entries[i].tile_offset = base;
            entries[i].tile_size = tile_params[i]->slice_data_size;
            entries[i].tile_row = tile_params[i]->tile_row;
            entries[i].tile_col = tile_params[i]->tile_column;
            base += tile_params[i]->slice_data_size;
        }
        req_ctrls[n_req_ctrls].id = V4L2_CID_STATELESS_AV1_TILE_GROUP_ENTRY;
        req_ctrls[n_req_ctrls].ptr = entries;
        req_ctrls[n_req_ctrls].size = sizeof(entries[0]) * n_tiles;
        n_req_ctrls++;
    }

    struct v4l2_ext_controls req_ctrls_batch = { 0 };
    req_ctrls_batch.controls = req_ctrls;
    req_ctrls_batch.count = n_req_ctrls;
    /* One S_EXT_CTRLS per frame. All ids are written every frame, so an
     * early return after this point can never leave a stale mixture on
     * the request. Failure is terminal — no per-control fallback. */
    if (v4l2sl_set_request_controls(request_fd, v4l2_fd, &req_ctrls_batch) < 0) {
        fprintf(stderr, "v4l2stateless: failed to set AV1 request controls\n");
        return VA_STATUS_ERROR_OPERATION_FAILED;
    }
```

Order FILM_GRAIN → FRAME → TILE_GROUP mirrors the old sequence (request-scoped storage is id-indexed, not order-sensitive).
